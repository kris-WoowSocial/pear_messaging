// Call Push Service - High-priority push messages for VoIP wake-up
// iOS: uses APNs VoIP push via PushKit (primary) — wakes even killed apps
// Android: uses FCM data-only message (primary)
const logger = require("../utils/logger")("CallPushService");
const admin = require("firebase-admin");
const dynamoDBFCMService = require("./dynamodb_fcm.service");
const redisService = require("./redis_message.service");
const usernameService = require("./username.service");
const apnsService = require("./apns.service");
const pendingVoipPushes = new Set();

/**
 * Send high-priority data-only FCM message to wake up device for incoming call
 * Uses data-only payload (no notification block) so OS delivers to background handler
 * 
 * @param {Object} params
 * @param {string} params.deviceId - Target device ID
 * @param {string} params.callId - Unique call ID
 * @param {string} params.callerId - Device ID of caller
 * @param {string} params.callerName - Display name of caller
 * @param {string} [params.callerAvatarUrl] - Avatar URL for caller
 * @param {number} params.timestamp - Call initiation timestamp
 * @returns {Promise<boolean>} - Success status
 */
async function sendIncomingCallPush({
  deviceId,
  callId,
  callerId,
  callerName,
  callerAvatarUrl = '',
  isVideo = false,
  timestamp = Date.now()
}) {
  // Resolve display name
  let displayName = callerName;
  if (callerId) {
    const resolved = usernameService.getDisplayName(callerId, deviceId);
    if (resolved && resolved !== callerId) {
      displayName = resolved;
    } else if (!displayName || displayName === 'Unknown') {
      displayName = `@user_${callerId.substring(callerId.length - 5)}`;
    }
  }
  if (!displayName || displayName === 'Unknown') {
    displayName = `@user_${(callerId || 'unknown').substring((callerId || 'unknown').length - 5)}`;
  }

  // iOS path: VoIP push is the ONLY reliable way to wake a killed iOS app.
  // If the device has a registered VoIP token, send APNs VoIP push and return.
  // FCM does NOT reliably wake killed iOS apps — PushKit VoIP push does.
  const voipToken = await redisService.getVoipToken(deviceId);
  if (voipToken) {
    logger.info("iOS device with VoIP token — sending APNs VoIP push (primary)", { deviceId, callId });
    return sendApnsVoipPush({ deviceId, callId, callerId, callerName: displayName, isVideo });
  }

  // Android / non-iOS fallback: FCM data-only message
  try {
    const fcmToken = await dynamoDBFCMService.getFCMToken(deviceId);
    if (!fcmToken) {
      logger.warn("No FCM token found for device", { deviceId, callId });
      return false;
    }

    // Build data-only payload — NO notification block (critical for background wake-up)
    // SECURITY: sdpOffer excluded — re-delivered via call_resync over authenticated socket
    const message = {
      data: {
        type: "incoming_call",
        callId: callId,
        callerId: callerId,
        callerName: displayName,
        ...(isVideo ? { isVideo: "true" } : {}),
        timestamp: timestamp.toString()
      },
      android: {
        priority: "high",
        ttl: 0,
        directBootOk: true
      },
      token: fcmToken
    };

    logger.info("Sending incoming call FCM push (Android)", {
      deviceId,
      callId,
      callerId,
      fcmTokenPrefix: fcmToken.substring(0, 20)
    });

    const response = await admin.messaging().send(message);
    logger.info("Incoming call FCM push sent", { deviceId, callId, messageId: response });
    return true;
  } catch (error) {
    logger.error("Failed to send incoming call FCM push", {
      deviceId,
      callId,
      error: error.message,
      code: error.code
    });

    if (error.code === 'messaging/registration-token-not-registered' ||
        error.code === 'messaging/invalid-argument') {
      await dynamoDBFCMService.deleteFCMToken(deviceId).catch(() => {});
      logger.info("Invalid FCM token deleted after push failure", { deviceId });
    }

    return false;
  }
}

/**
 * Attempt socket delivery with FCM fallback
 * 
 * @param {Object} params
 * @param {string} params.deviceId - Target device
 * @param {string} params.callId - Call ID
 * @param {string} params.callerId - Caller device ID
 * @param {string} params.callerName - Caller display name
 * @param {string} params.sdpOffer - SDP offer
 * @param {Function} params.socketDeliver - Function to attempt socket delivery
 * @param {number} params.timeoutMs - Timeout for socket delivery (default 1500ms)
 * @returns {Promise<{delivered: boolean, via: 'socket'|'fcm'|'failed'}>}
 */
async function deliverWithFallback({
  deviceId,
  callId,
  callerId,
  callerName,
  callerAvatarUrl = '',
  socketDeliver,
  timeoutMs = 1500
}) {
  const startTime = Date.now();
  
  logger.info("Starting call delivery with fallback", {
    deviceId,
    callId,
    callerId,
    timeoutMs
  });

  // Check if device is online via socket
  const isOnline = await isDeviceOnline(deviceId);
  
  if (!isOnline) {
    logger.info("Device offline via socket, immediately using FCM fallback", {
      deviceId,
      callId
    });
    
    const pushSent = await sendIncomingCallPush({
      deviceId,
      callId,
      callerId,
      callerName,
      callerAvatarUrl,
      timestamp: startTime
    });

    return {
      delivered: pushSent,
      via: pushSent ? 'fcm' : 'failed'
    };
  }

  // Device appears online, try socket delivery with timeout
  return new Promise((resolve) => {
    let resolved = false;
    let socketSuccess = false;

    // Set timeout for socket delivery
    const timeout = setTimeout(async () => {
      if (!resolved && !socketSuccess) {
        resolved = true;
        
        logger.info("Socket delivery timeout, triggering FCM fallback", {
          deviceId,
          callId,
          elapsedMs: Date.now() - startTime
        });

        const pushSent = await sendIncomingCallPush({
          deviceId,
          callId,
          callerId,
          callerName,
          callerAvatarUrl,
          timestamp: startTime
        });

        resolve({
          delivered: pushSent,
          via: pushSent ? 'fcm' : 'failed'
        });
      }
    }, timeoutMs);

    // Attempt socket delivery
    socketDeliver()
      .then((success) => {
        if (!resolved) {
          socketSuccess = success;
          
          if (success) {
            clearTimeout(timeout);
            resolved = true;
            
            logger.info("Call delivered via socket successfully", {
              deviceId,
              callId,
              elapsedMs: Date.now() - startTime
            });

            resolve({
              delivered: true,
              via: 'socket'
            });
          }
          // If socketDeliver returns false, let timeout continue to FCM fallback
        }
      })
      .catch((error) => {
        logger.error("Socket delivery error", {
          deviceId,
          callId,
          error: error.message
        });
        // Let timeout handle FCM fallback
      });
  });
}

/**
 * Send APNs VoIP push for iOS background wake
 * Requires APNs Authentication Key (.p8) configured in Firebase Console
 * 
 * @param {Object} params
 * @param {string} params.deviceId - Target device ID
 * @param {string} params.callId - Unique call ID
 * @param {string} params.callerId - Device ID of caller
 * @param {string} params.callerName - Display name of caller
 * @returns {Promise<boolean>} - Success status
 */
async function sendApnsVoipPush({
  deviceId,
  callId,
  callerId,
  callerName,
  isVideo = false
}) {
  const pushKey = `${deviceId}:${callId}`;
  if (pendingVoipPushes.has(pushKey)) {
    logger.debug("Skipping duplicate APNs VoIP push", { deviceId, callId });
    return false;
  }

  pendingVoipPushes.add(pushKey);
  try {
    // Get VoIP token for iOS device from Redis
  const voipToken = await redisService.getVoipToken(deviceId);
  if (!voipToken) {
    logger.debug("No VoIP token found for device (iOS may not be registered)", { deviceId, callId });
    return false;
  }

  // Send directly via APNs HTTP/2 — the ONLY way to deliver to a VoIP token.
  // Firebase Admin SDK (admin.messaging()) routes through FCM and CANNOT send
  // to APNs VoIP tokens. Direct APNs is how WhatsApp/Signal do this.
  // SECURITY: sdpOffer excluded — re-delivered via call_resync over authenticated socket
  const payload = {
    type: "incoming_call",
    callId: callId,
    callerId: callerId,
    callerName: callerName || "Unknown",
    isVideo: isVideo ? "true" : "false",
    timestamp: Date.now().toString()
  };

    const sent = await apnsService.sendVoipPush(voipToken, payload);
    if (sent) {
      logger.info("APNs VoIP push sent (direct HTTP/2)", { deviceId, callId, callerName });
    } else {
      logger.warn("APNs VoIP push failed — check APNS_KEY_ID/APNS_TEAM_ID/APNS_KEY env vars", { deviceId, callId });
    }
    return sent;
  } finally {
    pendingVoipPushes.delete(pushKey);
  }
}

// FCM token storage backed by DynamoDB
const { PutCommand, GetCommand, DeleteCommand } = require('@aws-sdk/lib-dynamodb');
const { dynamoDB, isConfigured } = require('../config/aws');
const logger = require('../utils/logger')('DynamoDBFCMService');

const TABLE = process.env.DYNAMODB_FCM_TABLE || 'fcm-tokens';
const FCM_TTL_SECONDS = 7 * 24 * 60 * 60; // 7 days

class DynamoDBFCMService {
  async storeFCMToken(deviceId, fcmToken) {
    if (!isConfigured()) {
      logger.warn('AWS not configured, FCM token not stored', { deviceId });
      return false;
    }

    try {
      const ttl = Math.floor(Date.now() / 1000) + FCM_TTL_SECONDS;

      await dynamoDB.send(new PutCommand({
        TableName: TABLE,
        Item: { deviceId, fcmToken, ttl },
      }));

      logger.debug('FCM token stored in DynamoDB', { deviceId });
      return true;
    } catch (error) {
      logger.error('Failed to store FCM token', { deviceId, error: error.message });
      return false;
    }
  }

  async getFCMToken(deviceId) {
    if (!isConfigured()) return null;

    try {
      const result = await dynamoDB.send(new GetCommand({
        TableName: TABLE,
        Key: { deviceId },
      }));

      return result.Item?.fcmToken || null;
    } catch (error) {
      logger.error('Failed to get FCM token', { deviceId, error: error.message });
      return null;
    }
  }

  async deleteFCMToken(deviceId) {
    if (!isConfigured()) return false;

    try {
      await dynamoDB.send(new DeleteCommand({
        TableName: TABLE,
        Key: { deviceId },
      }));

      logger.debug('FCM token deleted from DynamoDB', { deviceId });
      return true;
    } catch (error) {
      logger.error('Failed to delete FCM token', { deviceId, error: error.message });
      return false;
    }
  }
}

    private fun forwardToFlutter(remoteMessage: RemoteMessage) {
        try {
            val prefs = getSharedPreferences(
                "io.flutter.firebase.messaging.callback",
                Context.MODE_PRIVATE
            )
            val callbackHandle = prefs.getLong("callback_handle", 0L)

            if (callbackHandle == 0L) {
                Log.w(TAG, "No Flutter background callback registered. Message dropped.")
                return
            }

            // Never spin up a second engine during a ringing/active call — it
            // would compete for CPU/audio resources and run plugin init while
            // the user is on the call screen.
            if (VantCallManager.activeConnection != null) {
                Log.d(TAG, "Call in progress — deferring background forwarding")
                return
            }

            // FlutterLoader.startInitialization must run on the main thread —
            // onMessageReceived arrives on a worker thread, so hop over before
            // touching the engine. Everything downstream (MethodChannel invoke)
            // must be on the main thread anyway.
            Handler(Looper.getMainLooper()).post {

                            "messageId" to remoteMessage.messageId,

     * Incoming-call display name resolution (same order as the Dart
     * CallerNameResolver): non-placeholder payload name -> saved contact name
     * from FlutterSharedPreferences -> raw payload -> "Unknown". The wire's
     * callerName is often the sender's generated @user_xxxxx handle, which
     * must never win over a name the user actually saved.
     */
    private fun isPlaceholderCallerName(name: String?, callerId: String): Boolean {
        if (name.isNullOrBlank()) return true
        val t = name.trim()
        if (t == callerId || t.startsWith("@user_")) return true
        return t == "Unknown" || t == "Unknown User" ||
               t == "Incoming Call" || t == "Vant Call"
    }

    private fun savedContactName(callerId: String): String? {
        if (callerId.isEmpty()) return null
        return getSharedPreferences("FlutterSharedPreferences", Context.MODE_PRIVATE)
            .getString("flutter.chat_name_$callerId", null)
            ?.takeIf { it.isNotBlank() }
    }

    private fun resolveCallerName(callerId: String, payloadName: String?): String {
        if (!isPlaceholderCallerName(payloadName, callerId)) return payloadName!!.trim()
        val saved = savedContactName(callerId)
        if (!saved.isNullOrBlank()) return saved
        return payloadName?.trim()?.takeIf { it.isNotEmpty() } ?: "Unknown"
    }

    private fun handleIncomingCall(data: Map<String, String>) {
        val callerId = data["callerId"] ?: data["caller_id"] ?: ""
        val callerName = resolveCallerName(
            callerId, data["callerName"] ?: data["caller_name"]
        )
        val callId = data["callId"] ?: data["call_id"] ?: ""
        val roomId = data["roomId"] ?: data["room_id"] ?: ""
        val avatarUrl = data["avatarUrl"] ?: data["avatar_url"] ?: ""
        val isVideo = data["isVideo"] == "true" || data["is_video"] == "true"

            Log.w(TAG, "⚠️ Invalid incoming_call data: missing callId or callerId")
            return
        }

        VantCallManager.logIncomingCallDiagnostics(this)

        wakeUpDevice()

        if (isAppForeground() && isFlutterCallScreenVisible()) {
            Log.d(TAG, "📞 Flutter call screen already visible, skipping native UI")
            return
        }

        // If app is in foreground and unlocked, send broadcast to Flutter
        if (isAppForeground() && !isScreenLockedOrOff()) {
            Log.d(TAG, "📞 App foreground and unlocked, sending broadcast to Flutter")
            VantCallManager.startRinging(this, callerName, callerId, callId, roomId, avatarUrl, isVideo)
            sendFlutterCallBroadcast(data)
            return

        // Boot the Dart engine now so socket registration + resync + ICE
        // prefetch can complete during the ring — MainActivity attaches this
        // engine on Accept instead of cold-booting a second one.
        prewarmCallEngine(callId)

        // proactively fetch the offer via call_resync before Accept.

            .putString("flutter.ringing_caller_id", callerId)

            this, callerName, callerId, callId, roomId, avatarUrl, isVideo

            // answer/decline actions + full-screen intent is shown.

            VantCallManager.presentIncomingCall(this, callerName, callerId, callId, roomId, avatarUrl, isVideo)
